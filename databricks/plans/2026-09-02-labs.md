**Goal:** Every worksheet and answer notebook in `ai_cyber_bootcamp_2026` runs on the `bootcamp-labs` Databricks cluster against the `ai_cyber_bootcamp.course.data` volume, and still runs locally unchanged in intent, switching on the one `try: dbutils` seam cell the spec defines.

Verification is a one-off job run of each answer notebook on the existing cluster, recorded to a JSON run record and summarised in `DEPLOYMENT-NOTES.md`.

```python
SEAM_END = "# --- end data-loader ---"

_BODY = '''import os
try:
    dbutils  # defined only on Databricks
    DATA_HOME = os.environ.get("COURSE_DATA_HOME", "/Volumes/ai_cyber_bootcamp/course/data/")
    WORK_HOME = os.path.abspath("../work") + "/"

    WORK_HOME = DATA_HOME
    ON_DATABRICKS = False

def handoff(name):
    """Output of an earlier worksheet: the student's own copy if it exists, else the shipped one."""
    p = WORK_HOME + name
    return p if os.path.exists(p) else DATA_HOME + name
'''

_LLM_BODY = '''
COURSE_LLM_MODEL = (
    os.environ.get("COURSE_LLM_ENDPOINT", "databricks-deepseek-v4-flash-0731")
    if ON_DATABRICKS
    else os.environ.get("COURSE_LLM_MODEL", "claude-opus-4-8")
)


def make_client():
    """OpenAI-compatible chat client: Anthropic locally, the workspace endpoint on Databricks."""
    from openai import OpenAI
    if ON_DATABRICKS:
        ctx = dbutils.notebook.entry_point.getDbutils().notebook().getContext()
        return OpenAI(api_key=ctx.apiToken().get(),
                      base_url=f"{ctx.apiUrl().get()}/serving-endpoints")
    from dotenv import load_dotenv
    load_dotenv()
    return OpenAI(api_key=os.environ["ANTHROPIC_API_KEY"],
                  base_url="https://api.anthropic.com/v1/")


def make_llm(**kwargs):
    """LangChain chat model: ChatDatabricks on Databricks, ChatAnthropic locally."""
    if ON_DATABRICKS:
        from databricks_langchain import ChatDatabricks
        return ChatDatabricks(endpoint=COURSE_LLM_MODEL, **kwargs)
    from langchain_anthropic import ChatAnthropic
    return ChatAnthropic(model=COURSE_LLM_MODEL, **kwargs)
'''

SEAM_BASE = f"{SEAM_START}\n{_BODY}{SEAM_END}\n"
SEAM_LLM = f"{SEAM_START}\n{_BODY}{_LLM_BODY}{SEAM_END}\n"


def seam_text(with_llm: bool = False) -> str:
    """The seam cell source for one notebook."""
    return SEAM_LLM if with_llm else SEAM_BASE
```

```python
from dataclasses import dataclass

import nbformat


class EditError(Exception):
    """A notebook did not have the shape the substitution table says it has."""


@dataclass(frozen=True)
class Sub:
    """One exact substitution. `count` is how many times `old` must appear."""

    old: str
    new: str
    count: int = 1
    kind: str = "code"  # "code" or "markdown"


def load(path):
    with open(path, encoding="utf-8") as fh:
        return nbformat.read(fh, as_version=4)


def save(path, nb):
    with open(path, "w", encoding="utf-8") as fh:
        nbformat.write(nb, fh)


def _seam_indices(nb):
    return [i for i, c in enumerate(nb.cells)
            if c.cell_type == "code" and SEAM_START in c.source]


def upsert_seam(nb, with_llm: bool = False) -> bool:
    """Insert the seam cell, or replace an outdated one. True if the notebook changed."""
    wanted = seam_text(with_llm)
    found = _seam_indices(nb)
    if len(found) > 1:
        raise EditError(f"{len(found)} seam cells in one notebook: indices {found}")
    if found:
        cell = nb.cells[found[0]]
        if cell.source == wanted:
            return False
        cell.source = wanted
        return True
    # Insert after a leading run of markdown (the title block), else at the top.
    at = 0
    while at < len(nb.cells) and nb.cells[at].cell_type == "markdown":
        at += 1
    nb.cells.insert(at, nbformat.v4.new_code_cell(wanted))
    return True


def apply_subs(nb, subs) -> int:
    """Apply each Sub across the notebook. Idempotent; any surprise is an EditError."""
    applied = 0
    for sub in subs:
        cells = [c for c in nb.cells if c.cell_type == sub.kind]
        hits = sum(c.source.count(sub.old) for c in cells)
        if hits == 0:
            already = sum(c.source.count(sub.new) for c in cells)
            if already >= sub.count:
                continue  # already applied — idempotent re-run
            raise EditError(
                f"old text {sub.old!r} found 0 times, new text found "
                f"{already} times (expected {sub.count})"
            )
        if hits != sub.count:
            raise EditError(f"old text {sub.old!r} found {hits} times, expected {sub.count}")
        for c in cells:
            if sub.old in c.source:
                c.source = c.source.replace(sub.old, sub.new)
```

```python
def port(module: Module, root: Path, check_only: bool) -> list[str]:
    """Apply one module's seam and substitutions. Returns one report line per file."""
    lines = []
    for rel, extra in ((module.worksheet, module.ws_only), (module.answer, module.ans_only)):
        if rel is None:
            continue
        path = Path(root) / rel
        nb = load(path)
        changed = False
        if module.seam != "none":
            changed |= upsert_seam(nb, with_llm=(module.seam == "llm"))
        n = apply_subs(nb, list(module.subs) + list(extra))
        changed |= n > 0
        if changed and not check_only:
            save(path, nb)
        lines.append(f"{'WOULD-EDIT' if check_only and changed else 'ok        '} "
                     f"{rel}: seam={module.seam} subs={n}")
    return lines


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--check", action="store_true", help="report only, write nothing")
    ap.add_argument("--only", help="module number, e.g. 4.1")
    args = ap.parse_args()

    root = Path(__file__).resolve().parents[2]
    failures = 0
    for module in MODULES:
        if args.only and f"Worksheet_{args.only}_" not in module.worksheet:
            continue
        try:
            for line in port(module, root, args.check):
                print(line)
        except EditError as exc:
            failures += 1
            print(f"FAIL       {module.worksheet}: {exc}")
    print(f"{failures} module(s) failed")
    return 1 if failures else 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
#!/usr/bin/env python3
"""Gate for the Databricks port of the bootcamp notebooks.

CLI (from the course repo root):
    .venv/bin/python databricks/tools/check_notebook.py notebooks/*.ipynb answers/*.ipynb
    .venv/bin/python databricks/tools/check_notebook.py --all

Exit code is the number of violations (capped at 255). 0 means clean.

Rules:
  1. seam            the marker-commented data-loader cell appears exactly once
  2. data-paths      no '../data' literal in a code cell outside the seam cell
  3. pip             no !pip / %pip in a code cell outside the seam cell,
                     except Module 4.3's sanctioned TPOT install
  4. llm             no ChatAnthropic( / anthropic.Anthropic( / langchain_anthropic
                     import outside the seam cell
  5. pairs           every worksheet and its answer carry the same seam flavour
"""

import glob
import sys
from pathlib import Path

import nbformat

sys.path.insert(0, str(Path(__file__).resolve().parent))

from seam import SEAM_END, SEAM_START

# Module 4.3 only (spec Decision 12): TPOT needs an old setuptools, which no
# cluster library can express, so that one notebook installs it notebook-scoped.
TPOT_PIP = '%pip install "setuptools<58" tpot'
TPOT_NOTEBOOK = "Worksheet_4.3_Automate_It_All"

DATA_LITERALS = ("'../data", '"../data')
PIP_TOKENS = ("!pip", "%pip", "!conda", "%conda")
LLM_TOKENS = ("ChatAnthropic(", "anthropic.Anthropic(", "langchain_anthropic")


def load_notebook(path_str: str):
    path = Path(path_str)
    if not path.is_file():
        print(f"ERROR: not a file: {path_str}", file=sys.stderr)
        sys.exit(2)
    try:
        with open(path, encoding="utf-8") as fh:
            return nbformat.read(fh, as_version=4)
    except (ValueError, OSError) as exc:
        print(f"ERROR: cannot read {path_str}: {exc}", file=sys.stderr)
        sys.exit(2)


def _is_seam(cell) -> bool:
    return cell.cell_type == "code" and SEAM_START in cell.source


def _body_cells(nb):
    """Every code cell that is not the seam cell, with its index."""
    return [(i, c) for i, c in enumerate(nb.cells)
            if c.cell_type == "code" and not _is_seam(c)]


def check_seam(nb) -> list:
    seams = [i for i, c in enumerate(nb.cells) if _is_seam(c)]
    if not seams:
        return ["seam: no data-loader cell"]
    if len(seams) > 1:
        return [f"seam: {len(seams)} data-loader cells at indices {seams}"]
    if SEAM_END not in nb.cells[seams[0]].source:
        return [f"seam: cell {seams[0]} has a start marker but no end marker"]
    return []


def check_no_local_data_paths(nb) -> list:
    out = []
    for i, cell in _body_cells(nb):
        for line in cell.source.splitlines():
            if any(lit in line for lit in DATA_LITERALS):
                out.append(f"data-paths: cell {i}: {line.strip()[:100]}")
    return out


def check_no_pip(nb, name: str) -> list:
    allow_tpot = TPOT_NOTEBOOK in name
    out = []
    for i, cell in _body_cells(nb):
        for line in cell.source.splitlines():
            stripped = line.strip()
            if not any(stripped.startswith(tok) for tok in PIP_TOKENS):
                continue
            if allow_tpot and stripped == TPOT_PIP:
                continue
            out.append(f"pip: cell {i}: {stripped[:100]}")
    return out


def check_no_direct_llm(nb) -> list:
    out = []
    for i, cell in _body_cells(nb):
        for line in cell.source.splitlines():
            if line.lstrip().startswith("#"):
                continue
            for token in LLM_TOKENS:
                if token in line:
                    out.append(f"llm: cell {i}: {line.strip()[:100]}")
                    break
    return out


def check_pairs(root: Path) -> list:
    """Every worksheet and its answer must carry the same seam flavour."""
    sys.path.insert(0, str(Path(__file__).resolve().parent))
    from edits import MODULES

    out = []
    for module in MODULES:
        if not module.answer:
            continue
        flavours = []
        for rel in (module.worksheet, module.answer):
            nb = load_notebook(root / rel)
            seam = next((c.source for c in nb.cells if _is_seam(c)), "")
            flavours.append("llm" if "def make_llm(" in seam else
                            "base" if seam else "none")
        if flavours[0] != flavours[1]:
            out.append(f"pairs: {module.worksheet} is {flavours[0]} but its "
                       f"answer is {flavours[1]}")
    return out


def main() -> int:
    args = sys.argv[1:]
    root = Path(__file__).resolve().parents[2]
    if args == ["--all"]:
        paths = sorted(glob.glob(str(root / "notebooks" / "*.ipynb"))) + \
                sorted(glob.glob(str(root / "answers" / "*.ipynb")))
    elif args:
        paths = args
    else:
        print("Usage: check_notebook.py <notebook.ipynb> [more...] | --all")
        return 2

    total = 0
    for path_str in paths:
        nb = load_notebook(path_str)
        name = Path(path_str).name
        violations = (check_seam(nb)
                      + check_no_local_data_paths(nb)
                      + check_no_pip(nb, name)
                      + check_no_direct_llm(nb))
        if violations:
            print(f"\n[FAIL] {path_str}")
            for v in violations:
                print(f"  {v}")
            total += len(violations)
        else:
            print(f"[OK]   {path_str}")

    if args == ["--all"]:
        pair_violations = check_pairs(root)
        for v in pair_violations:
            print(f"  {v}")
        total += len(pair_violations)

    print(f"\n{total} violation(s)")
    return min(total, 255)


if __name__ == "__main__":
```

```python
#!/usr/bin/env python3
"""Mirror the course dataset onto the Unity Catalog volume.

Usage (from the course repo root):
    .venv/bin/python databricks/tools/upload_data.py [--dry-run] [--workers 16]

Uploads data/ as-is, plus the fingerprint images extracted from their zip, plus the
one ai_examples skill Module 11.2 reads. Idempotent: a remote file whose size already
matches the local one is skipped, so a re-run after a partial upload is cheap.
"""

import argparse
import os
import shutil
import sys
import tempfile
import time
import zipfile
from concurrent.futures import ThreadPoolExecutor, as_completed

from databricks.sdk import WorkspaceClient
from databricks.sdk.errors import NotFound

PROFILE = "gtk-workspace"
VOLUME = "/Volumes/ai_cyber_bootcamp/course/data"
FINGERPRINT_ZIP = "data/fingerprints/fingerprint_train_data.zip"
FINGERPRINT_DEST = "fingerprints/train_data"
SKILLS_SRC = "ai_examples/skills"
SKILLS_DEST = "skills"


def local_files(root: str) -> list:
    """(absolute path, size, destination-relative path) for every file under root."""
    out = []
    for base, _dirs, names in os.walk(root):
        for name in sorted(names):
            path = os.path.join(base, name)
            out.append((path, os.path.getsize(path), os.path.relpath(path, root)))
    return sorted(out)


def remote_sizes(w, prefix: str) -> dict:
    """Every file already on the volume under prefix, by relative path -> size."""
    sizes = {}
    stack = [f"{VOLUME}/{prefix}".rstrip("/")]
    while stack:
        here = stack.pop()
        try:
            entries = list(w.files.list_directory_contents(here))
        except NotFound:
            continue
        for entry in entries:
            if entry.is_directory:
                stack.append(entry.path)
            else:
                rel = entry.path[len(VOLUME) + 1:]
                sizes[rel] = entry.file_size
    return sizes


def upload_one(w, local: str, dest_rel: str) -> tuple:
    for attempt in range(3):
        try:
            with open(local, "rb") as fh:
                w.files.upload(f"{VOLUME}/{dest_rel}", fh, overwrite=True)
            return dest_rel, None
        except Exception as exc:  # noqa: BLE001 - report, retry, then give up
            if attempt == 2:
                return dest_rel, str(exc)
            time.sleep(2 * (attempt + 1))


def plan_uploads(scratch: str) -> list:
    """(local path, destination-relative path) for everything that should be on the volume."""
    jobs = [(p, rel) for p, _s, rel in local_files("data")]

    extracted = os.path.join(scratch, "train_data")
    os.makedirs(extracted, exist_ok=True)
    with zipfile.ZipFile(FINGERPRINT_ZIP) as zf:
        zf.extractall(scratch)
    # The zip contains a top-level train_data/ directory; tolerate either shape.
    root = extracted if os.listdir(extracted) else scratch
    jobs += [(p, f"{FINGERPRINT_DEST}/{rel}") for p, _s, rel in local_files(root)]

    jobs += [(p, f"{SKILLS_DEST}/{rel}") for p, _s, rel in local_files(SKILLS_SRC)]
    return jobs


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--dry-run", action="store_true")
    ap.add_argument("--workers", type=int, default=16)
    args = ap.parse_args()

    w = WorkspaceClient(profile=PROFILE)
    scratch = tempfile.mkdtemp(prefix="bootcamp_upload_")
    try:
        jobs = plan_uploads(scratch)
        already = remote_sizes(w, "")
        todo = [(p, rel) for p, rel in jobs
                if already.get(rel) != os.path.getsize(p)]
        total_bytes = sum(os.path.getsize(p) for p, _ in todo)
        print(f"PLAN {len(jobs)} files total, {len(todo)} to upload, "
              f"{total_bytes / 1e6:.1f} MB", flush=True)
        if args.dry_run:
            return 0

        done = errors = 0
        t0 = time.time()
        with ThreadPoolExecutor(max_workers=args.workers) as ex:
            futs = [ex.submit(upload_one, w, p, rel) for p, rel in todo]
            for fut in as_completed(futs):
                rel, err = fut.result()
                done += 1
                if err:
                    errors += 1
                    print(f"ERROR {rel}: {err}", flush=True)
                if done % 200 == 0 or done == len(todo):
                    print(f"PROGRESS {done}/{len(todo)} {errors} errors "
                          f"{time.time() - t0:.0f}s", flush=True)
        print(f"DONE {done} uploaded, {errors} errors, "
              f"{(time.time() - t0) / 60:.1f} min", flush=True)
        return 1 if errors else 0
    finally:
        shutil.rmtree(scratch, ignore_errors=True)


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
        pass

    print(f"{state:9} {seconds:7.1f}s  {name}", flush=True)
```

```python
def insert_before_seam(nb, source: str) -> bool:
    """Put `source` in its own code cell immediately before the seam cell.

    Used for the one sanctioned notebook-scoped install (Module 4.3): it has to run
    before the seam, because dbutils.library.restartPython() clears everything
    defined so far. Idempotent on the exact source text.
    """
    found = _seam_indices(nb)
    if not found:
        raise EditError("no seam cell to insert before")
    if any(c.cell_type == "code" and c.source == source for c in nb.cells):
        return False
    nb.cells.insert(found[0], nbformat.v4.new_code_cell(source))
    return True
```
